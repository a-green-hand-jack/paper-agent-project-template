% Paper-facing interface macros. Each one is listed, with its meaning, control
% cue and source, in .agents/memory/project/paper-interfaces.md. Reported
% numbers reach the text only through these macros. Call a macro without
% arguments with braces (\MethodName{}), so the following space survives.
% Reusable display helpers live in paper/style/paper-harness.sty.

% Interface: PaperTODO
% Meaning: a visible placeholder that keeps the paper compilable.
% Control: free; make release-check rejects active uses.
\providecommand{\PaperTODO}[1]{\textbf{[TODO: #1]}}

% Interface: PaperNote
% Meaning: a note between authors; shown only in the main build.
% Control: free; make release-check rejects active uses.
\providecommand{\PaperNote}[1]{\ifPaperNotes\textcolor{red}{[Note: #1]}\fi}

% Interface: PaperIdentity
% Meaning: declares, in authors.tex, a string the blind build must not contain.
% Control: locked; typesets nothing.
\providecommand{\PaperIdentity}[1]{}

% Interface: PaperTitle
% Meaning: the canonical title shared by every build.
% Control: bounded wording; a positioning change requires Human review.
\providecommand{\PaperTitle}{TODO Paper Title}

% Interface: MethodName
% Meaning: the approved paper-facing name of the proposed method or system.
% Control: locked meaning; changing the name requires a global consumer review.
\providecommand{\MethodName}{\PaperTODO{method name}}

% Interface: CoreTerm
% Meaning: the preferred recurring term for the paper's central concept.
% Control: bounded wording; semantic redefinition requires Human review.
\providecommand{\CoreTerm}{\PaperTODO{core term}}

% Interface: StateSymbol
% Meaning: the primary state or representation symbol used by the method.
% Control: bounded notation; a symbol change requires collision and consumer review.
\providecommand{\StateSymbol}{\ensuremath{\mathbf{s}}}

% Interface: MainResult
% Meaning: the main paper-facing result under the approved primary protocol.
% Control: bounded value; metric, split, aggregation, or checkpoint changes require Human review.
\providecommand{\MainResult}{\PaperTODO{main result}}

% Interface: MainResultUncertainty
% Meaning: the uncertainty paired with \MainResult under the same protocol.
% Control: bounded value; uncertainty type or aggregation changes require Human review.
\providecommand{\MainResultUncertainty}{\PaperTODO{main-result uncertainty}}
